    private static Investidor buscarInvestidor(String doc) {
        for (Investidor inv : listaInvestidores) {
            if (inv.getDocumento().equals(doc)) {
                return inv; // 🎯 Achou o investidor
            }
        }
        return null; // ❌ Não encontrado
